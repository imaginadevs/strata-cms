# 10 — Internationalization (i18n)

## Overview

Translatable fields are stored in a shadow table named `<collection>_translations` rather than creating duplicate records. A French translation of a blog post is a row in `posts_translations` linked to the original `posts` row — the original record always exists and holds the default-language content. Clients request content in a specific language via the `?lang=es` query parameter. The item repository merges the translation overlay onto the base record and falls back through the language chain if the requested translation is missing.

## Prerequisites

- Guide 06 completed (collection engine, fields with `Translatable` flag).
- Guide 03 completed (migrations applied, `languages` table exists).
- At least one collection with translatable fields.

## Step-by-Step

### 1. Data model overview

```
posts (base table)
├── id
├── tenant_id
├── title          ← default language content (e.g. English)
├── body           ← default language content
└── status         ← NOT translatable (same across all languages)

posts_translations (shadow table)
├── id
├── item_id        ← FK → posts.id
├── tenant_id
├── language       ← BCP 47 code, e.g. "es", "fr", "ar"
├── title          ← Spanish translation (nullable)
└── body           ← Spanish translation (nullable)
```

Non-translatable fields (status, dates, UUIDs) are always read from the base table. Translatable fields are replaced by the translation row value when `?lang=<code>` is provided and the translation exists.

```go
// In backend/internal/adapters/postgres/collection_ddl.go — update AddColumn:
func (r *CollectionRepository) AddColumn(ctx context.Context, _ uuid.UUID, f *collection.Field) error {
    col := buildColumnDDL(*f)
    if col == "" {
        return nil
    }

    // Add to main table
    _, err := r.db.Exec(ctx, fmt.Sprintf(
        "ALTER TABLE %s ADD COLUMN IF NOT EXISTS %s",
        pgIdentifier(f.CollectionName), col,
    ))
    if err != nil {
        return err
    }

    // Add to translations shadow table (nullable — translations may be partial)
    if f.Translatable {
        // Create the translations table if it doesn't exist yet
        if err := r.ensureTranslationsTable(ctx, f.CollectionName); err != nil {
            return err
        }
        // Add the nullable column to the shadow table
        _, err = r.db.Exec(ctx, fmt.Sprintf(
            "ALTER TABLE %s ADD COLUMN IF NOT EXISTS %s %s",
            pgIdentifier(f.CollectionName+"_translations"),
            pgIdentifier(f.Name),
            f.DBType,
        ))
    }
    return err
}

// ensureTranslationsTable creates the shadow table if it doesn't exist.
func (r *CollectionRepository) ensureTranslationsTable(ctx context.Context, collectionName string) error {
    tableName := pgIdentifier(collectionName + "_translations")
    ddl := fmt.Sprintf(`
        CREATE TABLE IF NOT EXISTS %s (
            id         UUID        PRIMARY KEY DEFAULT uuid_generate_v4(),
            item_id    UUID        NOT NULL REFERENCES %s(id) ON DELETE CASCADE,
            tenant_id  UUID        NOT NULL,
            language   VARCHAR(10) NOT NULL,
            created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
            updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
            UNIQUE (item_id, language)
        )`,
        tableName,
        pgIdentifier(collectionName),
    )
    _, err := r.db.Exec(ctx, ddl)
    if err != nil {
        return err
    }
    _, err = r.db.Exec(ctx, fmt.Sprintf(
        "CREATE INDEX IF NOT EXISTS idx_%s_translations_item ON %s (item_id, language)",
        collectionName, tableName,
    ))
    return err
}
```

### 4. Language fallback chain

The fallback order when requesting `?lang=es`:

```
1. Exact match: posts_translations WHERE language = 'es'
2. Base language: 'es' → try 'es-MX' or strip to 'es' (already tried)
3. Tenant default language (from tenants.default_language)
4. Application default ('en')
5. Base table content (always non-null)
```

Implement the fallback resolution:

```go
// backend/internal/adapters/postgres/language.go
package postgres

import (
    "context"
    "strings"

    "github.com/google/uuid"
    "github.com/jackc/pgx/v5/pgxpool"
)

type LanguageResolver struct {
    db *pgxpool.Pool
}

func NewLanguageResolver(db *pgxpool.Pool) *LanguageResolver {
    return &LanguageResolver{db: db}
}

// FallbackChain returns the ordered list of language codes to try.
// Example: requested="fr-CA", tenant default="en" → ["fr-CA", "fr", "en"]
func (r *LanguageResolver) FallbackChain(
    ctx context.Context,
    tenantID uuid.UUID,
    requestedLang string,
) []string {
    chain := []string{}

    if requestedLang != "" {
        chain = append(chain, requestedLang)
        // Strip region: "fr-CA" → "fr"
        if idx := strings.IndexByte(requestedLang, '-'); idx > 0 {
            base := requestedLang[:idx]
            if base != requestedLang {
                chain = append(chain, base)
            }
        }
    }

    // Add tenant default language
    var tenantDefault string
    _ = r.db.QueryRow(ctx,
        `SELECT default_language FROM tenants WHERE id = $1`,
        tenantID,
    ).Scan(&tenantDefault)

    if tenantDefault != "" && tenantDefault != requestedLang {
        chain = append(chain, tenantDefault)
    }

    // Always end with "en" as final fallback
    chain = append(chain, "en")

    // Deduplicate preserving order
    seen := map[string]bool{}
    result := chain[:0]
    for _, c := range chain {
        if !seen[c] {
            seen[c] = true
            result = append(result, c)
        }
    }
    return result
}
```

### 5. Applying translation overlay in the item repository

The key function is `applyTranslations` which fetches the best available translation and merges it onto the base item map:

```go
// backend/internal/adapters/postgres/item_translations.go
package postgres

import (
    "context"
    "fmt"

    "github.com/jackc/pgx/v5/pgxpool"
)

// applyTranslations overlays translated field values onto an item map.
// Only non-null translation values are applied — preserves base content if translation is partial.
func applyTranslations(
    ctx context.Context,
    db *pgxpool.Pool,
    collectionName string,
    itemID string,
    lang string,
    item map[string]any,
    fallbackChain []string,
) {
    translationsTable := collectionName + "_translations"

    for _, langCode := range fallbackChain {
        rows, err := db.Query(ctx, fmt.Sprintf(`
            SELECT * FROM %s
            WHERE  item_id = $1 AND language = $2`,
            pgIdentifier(translationsTable),
        ), itemID, langCode)
        if err != nil {
            break
        }

        if rows.Next() {
            translation, err := rowToMap(rows)
            rows.Close()
            if err != nil {
                break
            }
            // Merge non-null translation fields onto the base item
            for k, v := range translation {
                if k == "id" || k == "item_id" || k == "tenant_id" ||
                   k == "language" || k == "created_at" || k == "updated_at" {
                    continue
                }
                if v != nil {
                    item[k] = v
                }
            }
            // Add metadata about which language was applied
            item["_lang"] = langCode
            return
        }
        rows.Close()
    }

    // No translation found — item remains in base language
    item["_lang"] = collectionName // indicates no translation applied
}
```

Update `ItemRepository.GetByID()` to use the fallback chain:

```go
func (r *ItemRepository) GetByID(
    ctx context.Context,
    tenantID uuid.UUID,
    collectionName, id string,
    filter outbound.QueryFilter,
) (map[string]any, error) {
    // ... fetch base item ...

    if filter.Lang != "" && filter.Lang != "en" {
        chain := r.langResolver.FallbackChain(ctx, tenantID, filter.Lang)
        applyTranslations(ctx, r.db, collectionName, id, filter.Lang, item, chain)
    }

    return item, nil
}
```

### 6. Saving a translation

```go
// POST /api/items/{collection}/{id}/translations/{lang}
func (h *itemHandler) upsertTranslation(w http.ResponseWriter, r *http.Request) {
    tenantID, _ := httpMiddleware.TenantIDFromCtx(r.Context())
    collectionName := chi.URLParam(r, "collection")
    itemID := chi.URLParam(r, "id")
    lang := chi.URLParam(r, "lang")

    var data map[string]any
    if err := json.NewDecoder(r.Body).Decode(&data); err != nil {
        writeError(w, http.StatusBadRequest, "invalid body")
        return
    }

    // Only allow translatable fields — fetch field list and filter
    translatableFields, _ := h.fieldRepo.ListTranslatableByCollection(r.Context(), tenantID, collectionName)
    allowed := map[string]bool{}
    for _, f := range translatableFields {
        allowed[f.Name] = true
    }

    filteredData := map[string]any{}
    for k, v := range data {
        if allowed[k] {
            filteredData[k] = v
        }
    }

    if err := h.itemRepo.UpsertTranslation(r.Context(), tenantID, collectionName, itemID, lang, filteredData); err != nil {
        writeError(w, http.StatusInternalServerError, "error saving translation")
        return
    }

    writeJSON(w, http.StatusOK, filteredData)
}
```

Implementation in the item repository:

```go
func (r *ItemRepository) UpsertTranslation(
    ctx context.Context,
    tenantID uuid.UUID,
    collectionName, itemID, lang string,
    data map[string]any,
) error {
    if len(data) == 0 {
        return nil
    }

    translationsTable := pgIdentifier(collectionName + "_translations")

    cols := []string{`"item_id"`, `"tenant_id"`, `"language"`}
    placeholders := []string{"$1", "$2", "$3"}
    args := []any{itemID, tenantID, lang}
    updates := []string{}
    i := 4

    for col, val := range data {
        cols = append(cols, pgIdentifier(col))
        placeholders = append(placeholders, fmt.Sprintf("$%d", i))
        args = append(args, val)
        updates = append(updates, fmt.Sprintf("%s = $%d", pgIdentifier(col), i))
        i++
    }

    q := fmt.Sprintf(`
        INSERT INTO %s (%s) VALUES (%s)
        ON CONFLICT (item_id, language)
        DO UPDATE SET %s, updated_at = NOW()`,
        translationsTable,
        strings.Join(cols, ", "),
        strings.Join(placeholders, ", "),
        strings.Join(updates, ", "),
    )

    _, err := r.db.Exec(ctx, q, args...)
    return err
}
```

```bash
TOKEN="..."

# Create base post in English (default language)
ITEM=$(curl -s -X POST "http://localhost:8080/api/items/posts" \
  -H "Authorization: Bearer $TOKEN" \
  -H "X-Tenant-ID: $TENANT" \

# Response: {"id":"...","title":"Hola Mundo","body":"Contenido en español","status":"published","_lang":"es"}
```
